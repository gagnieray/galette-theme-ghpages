{%- comment -%} The orange cartouche galette.eu carries in its header, pointing at wherever the plugin's archives actually live. Both URLs are configurable, and the whole box disappears when a site declares neither. plugin: name: Fullcard # optional, defaults to site.title version: "2.2.1" archive: galette-plugin-fullcard # builds both URLs below on galette.eu release_url: … # overrides the derived stable URL nightly_url: … # overrides the derived nightly URL Galette plugin releases live on galette.eu rather than on GitHub — plugin-fullcard has no GitHub release at all — so `archive` plus `version` is the normal way to declare them, and the version then lives in exactly one place. {%- endcomment -%} {%- assign dl_name = site.plugin.name | default: site.title | default: site.github.repository_name -%} {%- assign dl_base = site.galette_download_url | default: 'https://galette.eu/download/plugins' -%} {%- assign release_url = site.plugin.release_url -%} {%- if release_url == nil and site.plugin.archive and site.plugin.version -%} {%- capture release_url -%}{{ dl_base }}/{{ site.plugin.archive }}-{{ site.plugin.version }}.tar.bz2{%- endcapture -%} {%- endif -%} {%- assign release_url = release_url | default: site.github.releases_url -%} {%- assign nightly_url = site.plugin.nightly_url -%} {%- if nightly_url == nil and site.plugin.archive -%} {%- capture nightly_url -%}{{ dl_base }}/{{ site.plugin.archive }}-dev.tar.bz2{%- endcapture -%} {%- endif -%} {%- if release_url or nightly_url -%}
    {%- if release_url %}
  • {{ dl_name }} {{ site.plugin.version | default: t_latest }}
    • {%- endif %} {%- if nightly_url %}
  • {{ dl_name }} nightly

{%- endif -%}
