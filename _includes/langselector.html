{%- comment -%} Lists every translation of the current page. Pages of a same logical page share a `ref` and differ by their `lang`, which the consuming site sets through path-scoped `defaults` — so Weblate never has to write that key itself. Two guards, both learned from a real site: a page without `ref` has no translations by definition, and `where: 'ref', nil` would otherwise match every other ref-less page; and a language is listed once even when several pages claim it, which is what a README.md rendered as the index alongside an index.md does. {%- endcomment -%} {%- assign cur_ref = page.ref | default: '' -%} {%- if cur_ref != '' -%} {%- assign siblings = site.pages | where: 'ref', cur_ref | sort: 'lang' -%} {%- assign distinct = '' -%} {%- for sibling in siblings -%} {%- assign sl = sibling.lang | default: 'en' -%} {%- assign marker = '|' | append: sl | append: '|' -%} {%- unless distinct contains marker -%} {%- assign distinct = distinct | append: marker -%} {%- endunless -%} {%- endfor -%} {%- assign distinct_count = distinct | split: '||' | size -%} {%- if distinct_count > 1 -%}
{%- assign emitted = '' -%} {%- for sibling in siblings -%} {%- assign sib_lang = sibling.lang | default: 'en' -%} {%- assign marker = '|' | append: sib_lang | append: '|' -%} {%- unless emitted contains marker -%} {%- assign emitted = emitted | append: marker -%} {%- assign sib_html_lang = sib_lang | replace: '_', '-' -%} {%- include lang-name.html lang=sib_lang -%} {%- if sib_lang == cur_lang -%} {{ lang_name }} {%- else -%} {{ lang_name }} {%- endif -%} {%- endunless -%} {%- endfor -%}
{%- endif -%} {%- endif -%}
